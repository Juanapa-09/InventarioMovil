6. Conexión entre dispositivos

Pasos de la guía: crea un proyecto en supabase.com, ejecuta el SQL de abajo, crea los usuarios y pega aquí los datos (Project Settings → API). Cada dispositivo lo configura una sola vez.

SQL para ejecutar una vez (Supabase → SQL Editor)

create table if not exists public.products(
  sku text primary key, name text not null, category text default '', location text default '',
  stock numeric, deleted boolean default false, updated_by text, updated_at timestamptz default now());
create table if not exists public.count_entries(
  id text primary key, session text not null, sku text not null,
  stage text not null check (stage in ('almacen','produccion')),
  qty numeric not null check (qty >= 0), user_name text, device text,
  counted_at timestamptz not null, voided boolean default false, updated_at timestamptz default now());
create index if not exists idx_entries_main on public.count_entries(session, sku, stage);
create index if not exists idx_entries_upd on public.count_entries(updated_at);
create index if not exists idx_products_upd on public.products(updated_at);

create or replace function public.touch() returns trigger language plpgsql as $$
begin new.updated_at := now(); return new; end $$;
create trigger trg_products_touch before insert or update on public.products for each row execute function public.touch();
create trigger trg_entries_touch before insert or update on public.count_entries for each row execute function public.touch();

alter table public.products enable row level security;
alter table public.count_entries enable row level security;
create policy "solo usuarios autenticados" on public.products for all to authenticated using (true) with check (true);
create policy "solo usuarios autenticados" on public.count_entries for all to authenticated using (true) with check (true);

alter publication supabase_realtime add table public.products, public.count_entries;
